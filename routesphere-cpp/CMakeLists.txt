cmake_minimum_required(VERSION 3.20)
project(statewalk_cpp VERSION 3.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

option(STATEWALK_BUILD_TESTS "Build the test executables" ON)
option(STATEWALK_BUILD_EXAMPLES "Build the examples" ON)
option(STATEWALK_WITH_SQLITE "SQLite persistence provider (needs libsqlite3-dev)" ON)
option(STATEWALK_WITH_FREESWITCH "Compile the in-process FreeSWITCH adapter (needs FreeSWITCH headers)" OFF)
set(FREESWITCH_INCLUDE_DIR "/usr/local/freeswitch/include/freeswitch" CACHE PATH "FreeSWITCH include dir (switch.h)")

find_package(Threads REQUIRED)

# ── the library: header-only ──────────────────────────────────────────
add_library(statewalk INTERFACE)
add_library(statewalk::statewalk ALIAS statewalk)
target_include_directories(statewalk INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(statewalk INTERFACE Threads::Threads)
target_compile_features(statewalk INTERFACE cxx_std_20)

if(STATEWALK_WITH_SQLITE)
  find_library(SQLITE3_LIB sqlite3)
  find_path(SQLITE3_INC sqlite3.h)
  if(SQLITE3_LIB AND SQLITE3_INC)
    add_library(statewalk_sqlite INTERFACE)
    add_library(statewalk::sqlite ALIAS statewalk_sqlite)
    target_include_directories(statewalk_sqlite INTERFACE ${SQLITE3_INC})
    target_link_libraries(statewalk_sqlite INTERFACE statewalk ${SQLITE3_LIB})
    set(STATEWALK_HAVE_SQLITE ON)
    message(STATUS "statewalk: SQLite provider enabled (${SQLITE3_LIB})")
  else()
    message(STATUS "statewalk: libsqlite3 not found — SQLite provider disabled")
  endif()
endif()

if(STATEWALK_WITH_FREESWITCH)
  if(EXISTS "${FREESWITCH_INCLUDE_DIR}/switch.h")
    add_library(statewalk_fs_check OBJECT src/fs_adapter_check.cpp)
    target_include_directories(statewalk_fs_check PRIVATE ${FREESWITCH_INCLUDE_DIR})
    target_link_libraries(statewalk_fs_check PRIVATE statewalk)
    message(STATUS "statewalk: FreeSWITCH adapter compile-check enabled (${FREESWITCH_INCLUDE_DIR})")
  else()
    message(WARNING "statewalk: switch.h not found under ${FREESWITCH_INCLUDE_DIR}")
  endif()
endif()

# ── tests: plain executables + ctest, no framework dependency ─────────
if(STATEWALK_BUILD_TESTS)
  enable_testing()
  function(sw_test name)
    add_executable(${name} tests/${name}.cpp)
    target_link_libraries(${name} PRIVATE statewalk)
    target_compile_options(${name} PRIVATE -Wall -Wextra)
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 120)
  endfunction()
  sw_test(state_map_test)
  sw_test(registry_core_test)
  sw_test(persistence_rehydration_test)
  sw_test(quota_test)
  sw_test(session_test)
  sw_test(channel_test)
  if(STATEWALK_HAVE_SQLITE)
    add_executable(sqlite_persistence_test tests/sqlite_persistence_test.cpp)
    target_link_libraries(sqlite_persistence_test PRIVATE statewalk_sqlite)
    target_compile_options(sqlite_persistence_test PRIVATE -Wall -Wextra)
    add_test(NAME sqlite_persistence_test COMMAND sqlite_persistence_test)
    set_tests_properties(sqlite_persistence_test PROPERTIES TIMEOUT 120)
  endif()
endif()

if(STATEWALK_BUILD_EXAMPLES)
  add_executable(call_fsm_example examples/call_fsm/main.cpp)
  target_link_libraries(call_fsm_example PRIVATE statewalk)
  target_compile_options(call_fsm_example PRIVATE -Wall -Wextra)
endif()
